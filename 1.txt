from flask import Flask, render_template, request
import pickle

app = Flask(__name__)

# Load anime data and similarity matrix
anime = pickle.load(open('anime.pkl', 'rb'))
similarity = pickle.load(open('similarity.pkl', 'rb'))

def recommend(anime_title):
    anime_title = anime_title.lower()
    if anime_title not in anime['title'].str.lower().values:
        return []

    index = anime[anime['title'].str.lower() == anime_title].index[0]
    distances = similarity[index]
    anime_list = sorted(list(enumerate(distances)), reverse=True, key=lambda x: x[1])[1:11]

    recommendations = []
    for i in anime_list:
        title = anime.iloc[i[0]].title
        synopsis = anime.iloc[i[0]].synopsis if 'synopsis' in anime.columns else "Synopsis not available"
        recommendations.append({'title': title, 'synopsis': synopsis})
    return recommendations

@app.route('/')
def home():
    anime_list = anime['title'].values
    return render_template('index.html', anime_list=anime_list)

@app.route('/recommend', methods=['POST'])
def recommend_route():
    selected_anime = request.form['anime']
    recommendations = recommend(selected_anime)
    return render_template('index.html',
                           anime_list=anime['title'].values,
                           selected_anime=selected_anime,
                           recommendations=recommendations)

if __name__ == '__main__':
    app.run(debug=True)
